# app_single_file.py from flask import Flask, request, render_template_string import xml.etree.ElementTree as ET from xml.dom import minidom import cgi # 用於對輸出進行HTML轉義，增加安全性 app = Flask(__name__) # 將整個 HTML 頁面結構定義為一個 Python 的多行字串 # 我們使用 Jinja2 的語法 {{ ... }} 和 {% ... %}，Flask 的 render_template_string 會處理它 HTML_TEMPLATE = """ 物流資料輸入 (純 Python)

GoodsList (Goods)
{% if xml_output %}

生成的 XML 結果:

{% endif %}
""" def pretty_print_xml(elem): """將 ElementTree 元素轉換為格式化（帶縮排）的字串""" rough_string = ET.tostring(elem, 'utf-8') reparsed = minidom.parseString(rough_string) return reparsed.toprettyxml(indent=" ") @app.route('/', methods=['GET', 'POST']) def index(): xml_output = "" if request.method == 'POST': # 接收表單資料 form_data = request.form # 建立 XML 根元素 root = ET.Element('LogisticsData') # 將主要欄位加入 XML main_fields = [ 'logisticsOrderCode', 'trackingNumber', 'waybillNumber', 'opTime', 'timeZone', 'transportType', 'fromPortCode', 'toPortCode', 'operator', 'operatorContact', 'opLocation', 'opCode', 'opRemark', 'trackingDescription', 'imgUrl', 'bigBagID', 'masterID', 'asnID', 'opTimeStamp', 'utcOffset', 'reRouteToCode' ] for field in main_fields: if form_data.get(field): ET.SubElement(root, field).text = form_data.get(field) # 處理巢狀結構: Parcel parcel_node = ET.SubElement(root, 'parcel') parcel_fields = [ 'weight', 'weightUnit', 'length', 'width', 'height', 'dimensionUnit', 'packingType', 'parcelQuantity', 'bigBagWeight', 'bigBagWeightUnit' ] for field in parcel_fields: if form_data.get(field): ET.SubElement(parcel_node, field).text = form_data.get(field) # 處理列表結構: goodsList (此範例僅處理一組商品，可擴充) goods_list_node = ET.SubElement(root, 'goodsList') goods_node = ET.SubElement(goods_list_node, 'goods') goods_fields = ['productID', 'hsCode', 'localName', 'clearanceMode'] for field in goods_fields: if form_data.get(field): ET.SubElement(goods_node, field).text = form_data.get(field) # 產生格式化的 XML 字串 # 使用 cgi.escape 避免 XML/HTML 注入問題 xml_output = cgi.escape(pretty_print_xml(root)) # 使用 render_template_string 來渲染儲存在變數中的 HTML # 並將 xml_output 變數傳遞給模板 return render_template_string(HTML_TEMPLATE, xml_output=xml_output) if __name__ == '__main__': app.run(debug=True)
